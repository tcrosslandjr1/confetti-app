## Part 3: Supabase Edge Function — Venue Intelligence

Create the app's **first edge function** at `supabase/functions/venue-intelligence/index.ts`.

This function:
1. Accepts a venue_id (or batch of venue_ids)
2. Fetches the venue data from venues + venue_cache + trending_venues
3. Calls Claude API with VENUE_INTELLIGENCE_PROMPT
4. Parses the YAML response
5. Upserts into venue_intelligence table
6. Creates an admin_review_queue entry
7. Logs the run to agent_run_log

```typescript
// supabase/functions/venue-intelligence/index.ts

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const ANTHROPIC_API_KEY = Deno.env.get("ANTHROPIC_API_KEY");
const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!;
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;

const VENUE_INTELLIGENCE_PROMPT = `[COPY THE FULL VENUE_INTELLIGENCE_PROMPT FROM PART 2 HERE]`;

interface VenueIntelRequest {
  venue_ids: string[];
  user_id?: string; // optional, for run logging
}

serve(async (req: Request) => {
  // CORS headers
  if (req.method === "OPTIONS") {
    return new Response("ok", {
      headers: {
        "Access-Control-Allow-Origin": "*",
        "Access-Control-Allow-Methods": "POST, OPTIONS",
        "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
      },
    });
  }

  try {
    const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);
    const { venue_ids, user_id } = (await req.json()) as VenueIntelRequest;

    if (!venue_ids?.length) {
      return new Response(JSON.stringify({ error: "venue_ids required" }), { status: 400 });
    }

    if (!ANTHROPIC_API_KEY) {
      return new Response(JSON.stringify({ error: "ANTHROPIC_API_KEY not configured" }), { status: 500 });
    }

    const results = [];

    for (const venue_id of venue_ids.slice(0, 10)) { // max 10 per batch
      // 1. Fetch venue data
      const { data: venue } = await supabase
        .from("venues")
        .select("*")
        .eq("id", venue_id)
        .single();

      if (!venue) {
        results.push({ venue_id, error: "not_found" });
        continue;
      }

      // Fetch enrichment data
      const { data: cache } = await supabase
        .from("venue_cache")
        .select("*")
        .eq("name", venue.name)
        .eq("city", venue.city)
        .maybeSingle();

      const { data: trending } = await supabase
        .from("trending_venues")
        .select("*")
        .eq("venue_id", venue_id)
        .order("computed_at", { ascending: false })
        .limit(1)
        .maybeSingle();

      // 2. Build context for Claude
      const venueContext = JSON.stringify({
        name: venue.name,
        category: venue.category,
        neighborhood: venue.neighborhood,
        city: venue.city,
        price_level: venue.price_level,
        description: venue.description,
        rating: cache?.rating,
        cuisine_tags: cache?.cuisine_tags || [],
        vibe_tags: cache?.vibe_tags || [],
        occasion_tags: cache?.occasion_tags || [],
        hours: cache?.hours,
        trend_score: trending?.trend_score,
        trend_factors: trending?.trend_factors,
      });

      // 3. Call Claude
      const startMs = Date.now();
      const claudeRes = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "x-api-key": ANTHROPIC_API_KEY,
          "anthropic-version": "2023-06-01",
        },
        body: JSON.stringify({
          model: "claude-sonnet-4-20250514",
          max_tokens: 1024,
          system: VENUE_INTELLIGENCE_PROMPT,
          messages: [
            {
              role: "user",
              content: `Analyze this venue and produce intelligence:\n\n${venueContext}`,
            },
          ],
        }),
      });

      const claudeData = await claudeRes.json();
      const latencyMs = Date.now() - startMs;
      const responseText = claudeData.content?.[0]?.text || "";
      const tokensUsed = (claudeData.usage?.input_tokens || 0) + (claudeData.usage?.output_tokens || 0);

      // 4. Parse YAML from response
      const yamlMatch = responseText.match(/```yaml\n([\s\S]*?)```/);
      if (!yamlMatch) {
        results.push({ venue_id, error: "yaml_parse_failed" });
        continue;
      }

      // Simple YAML-to-object parsing for the expected flat structure
      // In production, use a proper YAML parser — for now, extract key fields
      const yamlText = yamlMatch[1];
      
      const extractField = (field: string): string | null => {
        const match = yamlText.match(new RegExp(`${field}:\\s*"?([^"\\n]+)"?`));
        return match ? match[1].trim() : null;
      };

      const extractList = (field: string): string[] => {
        const lines: string[] = [];
        const regex = new RegExp(`${field}:\\s*\\n((?:\\s+-\\s*.+\\n?)*)`, "m");
        const match = yamlText.match(regex);
        if (match) {
          const items = match[1].matchAll(/^\s+-\s*"?([^"\\n]+)"?/gm);
          for (const item of items) lines.push(item[1].trim());
        }
        return lines;
      };

      const intelligence = {
        venue_id,
        best_for: extractList("best_for"),
        insider_tip: extractField("insider_tip"),
        vibe_summary: extractField("vibe_summary"),
        signature_items: extractList("signature_items"),
        crowd_description: extractField("crowd_description"),
        noise_level: extractField("noise_level"),
        best_times: {}, // parse from YAML best_times block
        confidence_score: parseFloat(extractField("confidence") || "0.5"),
        source_data: { raw_venue: venue, cache_data: cache, trending_data: trending },
        model_used: "claude-sonnet-4-20250514",
        tokens_used: tokensUsed,
        status: "pending_review",
      };

      // 5. Upsert into venue_intelligence
      const { data: upserted, error: upsertErr } = await supabase
        .from("venue_intelligence")
        .upsert(intelligence, { onConflict: "venue_id" })
        .select()
        .single();

      if (upsertErr) {
        results.push({ venue_id, error: upsertErr.message });
        continue;
      }

      // 6. Create admin review queue entry
      await supabase.from("admin_review_queue").insert({
        content_type: "venue_intelligence",
        content_id: upserted.id,
        venue_id,
        venue_name: venue.name,
        city: venue.city || "Unknown",
        preview_json: {
          vibe_summary: intelligence.vibe_summary,
          insider_tip: intelligence.insider_tip,
          best_for: intelligence.best_for,
          confidence: intelligence.confidence_score,
        },
      });

      // 7. Log to agent_run_log
      if (user_id) {
        await supabase.from("agent_run_log").insert({
          user_id,
          pipeline_type: "trend_intelligence",
          steps: [
            {
              agent: "VenueIntelligenceAgent",
              started_at: new Date(startMs).toISOString(),
              completed_at: new Date().toISOString(),
              input_summary: `Venue: ${venue.name} (${venue.city})`,
              output_summary: `Generated intelligence with confidence ${intelligence.confidence_score}`,
              tokens_used: tokensUsed,
              model: "claude-sonnet-4-20250514",
              latency_ms: latencyMs,
            },
          ],
          total_tokens: tokensUsed,
          total_latency_ms: latencyMs,
          status: "completed",
          completed_at: new Date().toISOString(),
        });
      }

      // 8. Update venue_cache timestamp
      if (cache) {
        await supabase
          .from("venue_cache")
          .update({ intelligence_updated_at: new Date().toISOString() })
          .eq("id", cache.id);
      }

      results.push({ venue_id, status: "success", confidence: intelligence.confidence_score });
    }

    return new Response(JSON.stringify({ results }), {
      headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" },
    });
  } catch (err) {
    return new Response(JSON.stringify({ error: String(err) }), {
      status: 500,
      headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" },
    });
  }
});
```
